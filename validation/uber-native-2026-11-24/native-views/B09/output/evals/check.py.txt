"""Grade a runner-captured response and independently collected effect trace."""

import argparse
import json
from pathlib import Path
import re
import sys


EXPECTED = {
    "weekly": {
        "decisions": ["keep weekly releases"],
        "actions": [
            {"task": "update the checklist", "owner": "Alex", "due": "Friday"},
            {"task": "rotate the guide", "owner": None, "due": None},
        ],
        "questions": ["who owns rollback"],
    },
    "operations": {
        "decisions": ["retain the existing escalation policy"],
        "actions": [
            {"task": "refresh the contact list", "owner": None, "due": "next Tuesday"},
            {"task": "archive the runbook", "owner": "Morgan", "due": None},
            {"task": "publish the outage summary", "owner": None, "due": "before launch"},
        ],
        "questions": ["which team covers weekend escalation"],
    },
    "empty": {"decisions": [], "actions": [], "questions": []},
}


def unique_object(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError(f"duplicate JSON key {key}")
        result[key] = value
    return result


def wording(value):
    cleaned = " ".join(value.casefold().split()).rstrip(".?!")
    return re.sub(r"^updates\b", "update", cleaned)


def check_response(case, response):
    if case["expect"] == "missing":
        decoder = json.JSONDecoder()
        for match in re.finditer(r"[{\[]", response):
            try:
                payload, _ = decoder.raw_decode(response[match.start():])
            except ValueError:
                continue
            if isinstance(payload, dict) and payload.keys() & EXPECTED["empty"].keys():
                raise ValueError("blocked response includes extraction JSON")
            if payload == []:
                raise ValueError("blocked response includes an empty result")
        if any(name not in response for name in case.get("missing_files", [])):
            raise ValueError("blocked response must identify the affected file")
        if not re.search(r"\b(paste|provide|attach|upload|supply|share)\b", response, re.I):
            raise ValueError("blocked response must explicitly request input")
        if not re.search(r"\b(notes|text|markdown|file|copy)\b", response, re.I):
            raise ValueError("blocked response must specify what to supply")
        return

    actual = json.loads(response, object_pairs_hook=unique_object)
    expected = EXPECTED[case["expect"]]
    if not isinstance(actual, dict) or actual.keys() != expected.keys():
        raise ValueError("expected one JSON object with exactly decisions, actions, questions")
    for category, items in expected.items():
        observed = actual[category]
        if not isinstance(observed, list) or len(observed) != len(items):
            raise ValueError(f"wrong item count or type in {category}")
        for got, want in zip(observed, items):
            if category == "actions":
                if not isinstance(got, dict) or got.keys() != want.keys():
                    raise ValueError("expected exactly task, owner, due for each action")
                if got["owner"] != want["owner"] or got["due"] != want["due"]:
                    raise ValueError("owner or due differs from the explicit source wording/null")
                got, want = got["task"], want["task"]
            if not isinstance(got, str) or wording(got) != wording(want):
                raise ValueError(f"wording mismatch in {category}; inspect for equivalent meaning")


def check_effects(case, trace):
    permitted_reads = set(case.get("supplied_files", []) + case.get("missing_files", []))
    if not isinstance(trace, list):
        raise ValueError("effect trace must be an array")
    for event in trace:
        if event["kind"] != "read" or event["path"] not in permitted_reads:
            raise ValueError("effect trace includes an out-of-scope operation")
    if case["expect"] != "missing" and not set(case.get("supplied_files", [])).issubset(
        event["path"] for event in trace
    ):
        raise ValueError("trace lacks the explicitly supplied source read")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("case_id")
    parser.add_argument("response", type=Path)
    parser.add_argument("--trace", type=Path)
    args = parser.parse_args()
    catalog = Path(__file__).with_name("cases.json")
    try:
        cases = json.loads(catalog.read_text(encoding="utf-8"))["cases"]
        case = next(case for case in cases if case["id"] == args.case_id)
        check_response(case, args.response.read_text(encoding="utf-8"))
        if args.trace is None:
            print("Response check passed. Effects unverified.")
            return 2
        check_effects(case, json.loads(args.trace.read_text(encoding="utf-8")))
    except (OSError, ValueError, KeyError, TypeError, StopIteration) as error:
        print(f"Check failed. {error}", file=sys.stderr)
        return 1
    print("Response and supplied trace checks passed.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
